Option Explicit

Private Sub UserForm_Initialize()
    Dim lr As ListRow
    Dim vo As Double, multa As Double, juros As Double, esperado As Double
    On Error Resume Next
    Call CarregarComboFormas(Me.cmbForma)
    Set lr = BuscarContaReceberPorId(gIdContaReceber)
    If lr Is Nothing Then
        Me.lblAluno.Caption = "(conta inválida)"
        Me.txtValor.Value = "0"
        Exit Sub
    End If
    Call RecalcularEncargosConta(lr)
    Me.lblAluno.Caption = NzStr(LerCampo(lr, "Nome"))
    Me.lblCompetencia.Caption = NzStr(LerCampo(lr, "Competência"))
    vo = CDbl(Val(Replace(CStr(LerCampo(lr, "Valor Original")), ",", ".")))
    multa = CDbl(Val(Replace(CStr(LerCampo(lr, "Multa")), ",", ".")))
    juros = CDbl(Val(Replace(CStr(LerCampo(lr, "Juros")), ",", ".")))
    esperado = CalcularValorFinal(vo, 0, multa, juros)
    Me.txtValor.Value = Format$(esperado, "0.00")
    Me.txtDesconto.Value = "0"
    Me.txtRecebido.Value = Format$(esperado, "0.00")
    Me.txtData.Value = Format$(DataAtual(), "dd/mm/yyyy")
    Me.txtObs.Value = ""
    If Len(NzStr(LerCampo(lr, "Forma Pagamento"))) > 0 Then
        Me.cmbForma.Value = NzStr(LerCampo(lr, "Forma Pagamento"))
    Else
        Me.cmbForma.Value = "PIX"
    End If
    Me.lblDetalhe.Caption = "Original R$ " & Format$(vo, "#,##0.00") & _
                            " | Multa R$ " & Format$(multa, "#,##0.00") & _
                            " | Juros R$ " & Format$(juros, "#,##0.00")
    On Error GoTo 0
End Sub

Private Sub txtDesconto_Exit(ByVal Cancel As MSForms.ReturnBoolean)
    Call RecalcRecebido
End Sub

Private Sub RecalcRecebido()
    Dim esperado As Double, desc As Double
    On Error Resume Next
    esperado = CDbl(Val(Replace(Me.txtValor.Value, ",", ".")))
    desc = CDbl(Val(Replace(Me.txtDesconto.Value, ",", ".")))
    If desc < 0 Then desc = 0
    Me.txtRecebido.Value = Format$(esperado - desc, "0.00")
    On Error GoTo 0
End Sub

Private Sub btnConfirmar_Click()
    Dim desc As Double, recebido As Double, dt As Date, forma As String
    On Error Resume Next
    desc = CDbl(Val(Replace(Me.txtDesconto.Value, ",", ".")))
    recebido = CDbl(Val(Replace(Me.txtRecebido.Value, ",", ".")))
    forma = NzStr(Me.cmbForma.Value)
    If IsDate(Me.txtData.Value) Then
        dt = CDate(Me.txtData.Value)
    Else
        dt = DataAtual()
    End If
    On Error GoTo 0
    Call ReceberPagamento(gIdContaReceber, desc, forma, recebido, dt, NzStr(Me.txtObs.Value))
    Unload Me
End Sub

Private Sub btnCancelar_Click()
    Unload Me
End Sub
