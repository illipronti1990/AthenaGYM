Option Explicit

Private Sub UserForm_Initialize()
    Me.Caption = "ATHENAS GYM — Login"
    Me.txtUsuario.Value = ""
    Me.txtSenha.Value = ""
    Me.txtSenha.PasswordChar = "*"
    Me.chkMostrarSenha.Value = False
    Me.lblMensagem.Caption = ""
    On Error Resume Next
    Me.txtUsuario.SetFocus
    On Error GoTo 0
End Sub

Private Sub chkMostrarSenha_Click()
    If Me.chkMostrarSenha.Value Then
        Me.txtSenha.PasswordChar = ""
    Else
        Me.txtSenha.PasswordChar = "*"
    End If
End Sub

Private Sub btnEntrar_Click()
    Dim usuario As String, senha As String
    usuario = Trim$(CStr(Me.txtUsuario.Value))
    senha = CStr(Me.txtSenha.Value)

    Me.lblMensagem.Caption = ""

    If Len(usuario) = 0 Or Len(senha) = 0 Then
        Me.lblMensagem.Caption = "Informe usuário e senha."
        Exit Sub
    End If

    If ValidarLogin(usuario, senha) Then
        RegistrarLogin
        MsgBox "Bem-vindo, " & NomeUsuario, vbInformation, "ATHENAS GYM"
        Unload Me
        ConcluirLogin
    Else
        Me.lblMensagem.Caption = "Usuário ou senha inválidos."
        MsgBox "Usuário ou senha inválidos.", vbCritical, "ATHENAS GYM"
        Me.txtSenha.Value = ""
        On Error Resume Next
        Me.txtSenha.SetFocus
        On Error GoTo 0
    End If
End Sub

Private Sub btnCancelar_Click()
    Unload Me
End Sub
